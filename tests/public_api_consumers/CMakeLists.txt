cmake_minimum_required(VERSION 3.24)
project(QCAEPublicConsumers LANGUAGES CXX)
if(NOT QCAE_SOURCE_ROOT)
  get_filename_component(QCAE_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
# This is an external consumer: never add product include paths here. Every
# header must compile using only the linked target's public usage requirements.
set(BUILD_TESTING OFF CACHE BOOL "Disable nested product tests" FORCE)
add_subdirectory("${QCAE_SOURCE_ROOT}" qcae)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${QCAE_SOURCE_ROOT}/modules/targets.json")
file(READ "${QCAE_SOURCE_ROOT}/modules/targets.json" manifest)
string(JSON count LENGTH "${manifest}" targets)
math(EXPR last "${count} - 1")
set(consumers)
set(header_count 0)
foreach(index RANGE ${last})
  string(JSON target GET "${manifest}" targets ${index} target)
  if(NOT TARGET ${target})
    continue()
  endif()
  string(JSON headers LENGTH "${manifest}" targets ${index} public_headers)
  if(headers EQUAL 0)
    continue()
  endif()
  math(EXPR last_header "${headers} - 1")
  foreach(header_index RANGE ${last_header})
    string(JSON path GET "${manifest}" targets ${index} public_headers ${header_index})
    string(REGEX REPLACE ".*/include/" "" include "${path}")
    string(MAKE_C_IDENTIFIER "consumer_${target}_${include}" consumer)
    set(source "${CMAKE_CURRENT_BINARY_DIR}/${consumer}.cpp")
    file(WRITE "${source}" "#include \"${include}\"\nint main() { return 0; }\n")
    add_executable(${consumer} EXCLUDE_FROM_ALL "${source}")
    target_link_libraries(${consumer} PRIVATE ${target})
    target_compile_features(${consumer} PRIVATE cxx_std_20)
    qcae_warnings(${consumer})
    list(APPEND consumers ${consumer})
    math(EXPR header_count "${header_count} + 1")
  endforeach()
endforeach()
# The operation catalog is generated into the build tree, so it is not a
# handwritten public_headers entry in the source ownership manifest.
set(catalog_source "${CMAKE_CURRENT_BINARY_DIR}/consumer_generated_operations.cpp")
file(WRITE "${catalog_source}" "#include \"qcae/operations.hpp\"\nint main() { return qcae::operation_catalog.empty() ? 1 : 0; }\n")
add_executable(consumer_generated_operations EXCLUDE_FROM_ALL "${catalog_source}")
target_link_libraries(consumer_generated_operations PRIVATE qcae_contracts)
qcae_warnings(consumer_generated_operations)
list(APPEND consumers consumer_generated_operations)
math(EXPR header_count "${header_count} + 1")
# Generated typed input and legacy catalog headers must also compose in one translation unit.
set(typed_source "${CMAKE_CURRENT_BINARY_DIR}/consumer_operation_contracts.cpp")
file(WRITE "${typed_source}" "#include \"qcae/operations.hpp\"\n#include \"qcae/operation_inputs.hpp\"\nint main() { return qcae::operation_catalog.empty() ? 1 : 0; }\n")
add_executable(consumer_operation_contracts EXCLUDE_FROM_ALL "${typed_source}")
target_link_libraries(consumer_operation_contracts PRIVATE qcae_operations)
qcae_warnings(consumer_operation_contracts)
list(APPEND consumers consumer_operation_contracts)
math(EXPR header_count "${header_count} + 1")
add_custom_target(qcae_public_consumers ALL DEPENDS ${consumers})
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/public-consumer-count.txt" "${header_count}\n")
message(STATUS "Public API gate: ${header_count} independently compiled header consumers")
